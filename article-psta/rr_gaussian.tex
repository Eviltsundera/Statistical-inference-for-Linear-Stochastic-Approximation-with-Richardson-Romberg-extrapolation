\section{Гауссовская аппроксимация для усреднения Поляка-Рупперта с экстраполяцией Ричардсона-Ромберга}\label{sec:theory}

В этой секции запись $x\lesssim_{\log{n}}y$ означает, что
$x\leq C(\log n)^q y$ для некоторых не зависящих от $n$ и $\alpha$
констант $C,q\geq0$. Иными словами, такое неравенство выполнено с точностью
до постоянного и логарифмического множителей; $C$ может зависеть от
фиксированных параметров задачи.

\subsection*{Контроль остаточного члена в разложении ошибки для итераций Ричардсона--Ромберга}

Для начального распределения $\xi$ цепи $(Z_k)_{k\geq0}$ обозначим через
$\PE_\xi$ соответствующее математическое ожидание, то есть $Z_0\sim\xi$.

\begingroup
\emergencystretch=2em
Для $p\geq2$ через $\alpha_p^{\rm L}$ обозначим верхнюю границу шага из
\cite[Теорема~2]{Levin2026}. Разложения (26)--(27), оценка (25) и её точная версия (87) в приложении этой
работы дают представление
\par
\endgroup
\begin{equation}\label{eq:levin-decomposition}
 \bA\bigl(\btheta_n^{(\alpha,\RR)}-\thetalim\bigr)
 =-\frac2n\sum_{k=n/2}^{n-1}\funcnoise{Z_{k+1}}
   +\scrE_{n,p,\alpha}^{\RR},
\end{equation}
где при $\alpha\leq\alpha_p^{\rm L}$ для любого начального распределения
$\xi$,
\begin{equation}\label{eq:levin-remainder}
 \left\{\PE_\xi\left[
  \norm{\scrE_{n,p,\alpha}^{\RR}}^p
 \right]\right\}^{1/p}
 \leq C_{\rm L}(\taumix,d)\,r_{n,p,\alpha},
\end{equation}
и, с $L_\alpha=\log(1/(\alpha a))$,
\begin{align}\label{eq:levin-rate}
 r_{n,p,\alpha}
 &=p\log(2p)n^{-3/4}
 +p^3(\alpha n)^{-1/2}L_\alpha^{1/2}n^{-1/2}\notag\\
 &\quad+p^{3/2}\alpha^{1/2}n^{-1/2}
 +p^{7/2}\alpha^{3/2}L_\alpha^{3/2}\notag\\
 &\quad
 +p^{7/2}n^{-1}\alpha^{1/2}L_\alpha^{3/2}\notag\\
 &\quad
 +\frac{\norm{\thetainit-\thetalim}}{\alpha n}
   \exp\!\left(-\frac{\alpha an}{24}\right).
\end{align}
Здесь и далее константы могут зависеть от фиксированных параметров задачи,
но не от $n$, $p$ и $\alpha$. Формула \eqref{eq:levin-remainder} ---
единственная необходимая далее оценка момента высокого порядка.

\subsection*{Основной результат}

Для $u\in\sphere^{d-1}$ обозначим нормированную ошибку через
\[
 T_{n,\alpha}^{\RR}(u)
 :=\frac{\sqrt{n/2}\,u^\transpose
 (\btheta_n^{(\alpha,\RR)}-\thetalim)}{\sigma(u)}.
\]
Наша цель --- получить неасимптотическую оценку расстояния
\[
 \dK\bigl(T_{n,\alpha}^{\RR}(u),\gauss(0,1)\bigr).
\]

Сначала поясним схему доказательства. Положим $m=n/2$ и
\[
 v:=\bA^{-\transpose}u,
 \qquad g(z):=v^\transpose\funcnoise{z}.
\]
Поскольку $v^\transpose\bA=u^\transpose$, разложение
\eqref{eq:levin-decomposition} представляет нормированную ошибку в виде
\begin{equation}
  \label{eq:T_RR}
 T_{n,\alpha}^{\RR}(u)
 =-\frac1{\sqrt m\,\sigma(u)}
   \sum_{k=m}^{n-1}g(Z_{k+1})
 +\frac{\sqrt m}{\sigma(u)}
   v^\transpose\scrE_{n,p,\alpha}^{\RR}.
\end{equation}
Первое слагаемое является ведущей линейной частью, а второе --- остатком,
для которого доступна оценка момента высокого порядка
\eqref{eq:levin-remainder}.

Ведущую сумму, в свою очередь, можно разложить с помощью уравнения Пуассона.
Пусть $\widehat g$ --- ограниченное решение
$\widehat g-\MKQ\widehat g=g$. Тогда
\begin{equation}
  \label{eq:poisson}
 \sum_{k=m}^{n-1}g(Z_{k+1})
 =\sum_{k=m}^{n-1}
   \{\widehat g(Z_{k+1})-\MKQ\widehat g(Z_k)\}
 +\MKQ\widehat g(Z_m)-\MKQ\widehat g(Z_n).
\end{equation}
Сумма в правой части состоит из мартингальных разностей; её нормальная
аппроксимация устанавливается в Лемме~\ref{lem:martingale-gaussian}. Крайние
члены пуассоновского разложения объединяются с
$\scrE_{n,p,\alpha}^{\RR}$ в остаточную случайную величину. Её момент
контролируется с помощью \eqref{eq:levin-remainder} и ограниченности
$\widehat g$. Для их объединения используем неравенство
\cite[(11)]{Samsonov2025}: для любых вещественных случайных величин $X$, $Y$
и любого $p\geq1$
\begin{equation}\label{eq:smoothing-inequality}
 \dK\!\left(X+Y,\gauss(0,1)\right)
 \leq \dK\!\left(X,\gauss(0,1)\right)
 +2\left\{\PE\left[\abs{Y}^p\right]\right\}^{1/(p+1)}.
\end{equation}
Таким образом, достаточно отдельно оценить расстояние Колмогорова для
мартингальной части и моменты остаточной случайной величины.

Чтобы компактно записать зависимость от шага, положим
\begin{align}\label{eq:alpha-n-rate}
 r_{n,\alpha}
 &: =n^{-1/4}
 +\frac{L_\alpha^{1/2}}{\sqrt{\alpha n}}
 +\sqrt\alpha
 +\sqrt n\,\alpha^{3/2}L_\alpha^{3/2}\notag\\
 &\quad
 +\frac{\sqrt\alpha\,L_\alpha^{3/2}}{\sqrt n}
 +\frac1{\alpha\sqrt n}\exp(-c_{\rm tr}\alpha n),
\end{align}
где $c_{\rm tr}>0$ зависит только от фиксированных параметров устойчивости.
Кроме того, обозначим
\[
 \alpha_n^{\rm L}:=
 \alpha_{\max\{2,\ceil{\log n}\}}^{\rm L}.
\]

\begin{lemma}[Мартингальная аппроксимация]
\label{lem:martingale-gaussian}
Пусть выполнены условия 1--3, $\noisecovmarkov\succ0$, $n$ чётно,
$n\geq2\taumix$ и $u\in\sphere^{d-1}$. Определим
\begin{equation}\label{eq:martingale-part}
 X_n(u):=-\frac1{\sqrt{n/2}\,\sigma(u)}
 \sum_{k=n/2}^{n-1}
 \{\widehat g(Z_{k+1})-\MKQ\widehat g(Z_k)\}.
\end{equation}
Тогда существуют не зависящие от $n$, $p$ и $u$ константы $C_1,C_2$ такие,
что для
\begin{align}\label{eq:BM-definition}
 \mathcal B_{\rm M}(n,p)
 &:=\frac{C_1\,
 p^{(3p+1)/(2(2p+1))}}
 {(n/2)^{p/(2(2p+1))}}\notag\\
 &\quad+C_2\left\{
 \frac{\log n}{\sqrt{n/2}}
 +\frac{p}{(n/2)^{p/(2p+1)}}\right\}
\end{align}
выполнено неравенство
\begin{equation}\label{eq:additive-gaussian}
 \dK\!\left(X_n(u),\gauss(0,1)\right)
 \leq\mathcal B_{\rm M}(n,p).
\end{equation}
\end{lemma}

\begin{proof}
Из \eqref{eq:sigmainfty} и \eqref{eq:directional-variance} следует
\begin{equation}\label{eq:projected-variance}
 \sigma^2(u)=v^\transpose\noisecovmarkov v.
\end{equation}
Таким образом, $\sigma(u)>0$, а предельная дисперсия $g(Z_k)$ равна
$\sigma^2(u)$. Кроме того,
\begin{equation}\label{eq:direction-uniform-bounds}
 \frac{\norm{v}}{\sigma(u)}\leq\Const{\varepsilon},
 \qquad
 \frac{\norm{g}[\infty]}{\sigma(u)}
 \leq \Const{\varepsilon}\supconsteps.
\end{equation}
Условие равномерной геометрической эргодичности обеспечивает оценку
\[
 \norm{\widehat g}[\infty]
 \lesssim\taumix\norm{g}[\infty].
\]

Положим $m=n/2$ и для $j=0,\ldots,m-1$ обозначим
\[
 D_{j+1}:=\widehat g(Z_{m+j+1})-\MKQ\widehat g(Z_{m+j}),
 \qquad
 M_m:=\sum_{j=0}^{m-1}D_{j+1}.
\]
Относительно фильтрации
$\filtr_{m+j}=\sigma(Z_0,\ldots,Z_{m+j})$ величины $D_{j+1}$ являются
мартингальными разностями. Если
\[
 h(z):=\MKQ(\widehat g^2)(z)
       -\{\MKQ\widehat g(z)\}^2,
\]
то
\[
 \PE_\xi[D_{j+1}^2\mid\filtr_{m+j}]=h(Z_{m+j}),
 \qquad
 \langle M\rangle_m=\sum_{j=0}^{m-1}h(Z_{m+j}).
\]
При стационарном начальном распределении из уравнения Пуассона следует
тождество
\[
 \PE_\pi[h(Z)]
 =\PE_\pi\!\left[
   \{\widehat g(Z_1)-\MKQ\widehat g(Z_0)\}^2
  \right]
 =\sigma^2(u).
\]
Следовательно, $m\sigma^2(u)$ является детерминированным центром
квадратической характеристики $M_m$.

Начальным распределением сдвинутой цепи $(Z_{m+j})_{j\geq0}$ служит закон
$Z_m$ при $Z_0\sim\xi$. Оценка из \cite[Лемма~22]{Samsonov2025} равномерна
по начальному распределению, поэтому её применение к сдвинутой
цепи даёт
\begin{equation}\label{eq:quadratic-characteristic-moment}
 \left\{\PE_\xi\left[
  \abs{\langle M\rangle_m-m\sigma^2(u)}^p
 \right]\right\}^{1/p}
 \leq 13^2\norm{g}[\infty]^2p^{1/2}\taumix^{5/2}\sqrt m.
\end{equation}
Кроме того,
\[
 \abs{D_{j+1}}\leq2\norm{\widehat g}[\infty]
 \lesssim\taumix\norm{g}[\infty].
\]
Применим \cite[Лемма~21]{Samsonov2025} с
$s_m^2=m\sigma^2(u)$. Поскольку
\[
 X_n(u)=-\frac{M_m}{\sqrt m\,\sigma(u)},
\]
подстановка \eqref{eq:quadratic-characteristic-moment} и оценки приращений в
эту Лемму вместе с \eqref{eq:direction-uniform-bounds} даёт
\eqref{eq:additive-gaussian}.
\end{proof}

\begin{lemma}[Остаточный член]
\label{lem:rr-remainder}
Пусть выполнены условия Леммы~\ref{lem:martingale-gaussian}. Пусть также
$p\geq2$ и $0<\alpha\leq\alpha_p^{\rm L}$. Тогда существует не зависящая
от $n$, $p$, $\alpha$ и $u$ константа $C_{\rm P}>0$ такая, что для
\begin{equation}\label{eq:RR-normalized-remainder}
 \mathcal R_{n,p,\alpha}
 :=C_{\rm P}\,
 \frac{\Const{\varepsilon}\taumix\supconsteps}
 {\sqrt{n/2}}
 +\Const{\varepsilon}C_{\rm L}(\taumix,d)
  \sqrt{n/2}\,r_{n,p,\alpha}
\end{equation}
выполнено неравенство
\begin{equation}\label{eq:intermediate-gaussian}
 \dK\!\left(T_{n,\alpha}^{\RR}(u),\gauss(0,1)\right)
 \leq \mathcal B_{\rm M}(n,p)
 +2\mathcal R_{n,p,\alpha}^{p/(p+1)} \eqsp,
\end{equation}
где $\mathcal B_{\rm M}(n, p)$ определен в \eqref{eq:BM-definition}.
\end{lemma}

\begin{proof}
Из \eqref{eq:T_RR} и \eqref{eq:poisson} следует разложение
\begin{equation}\label{eq:normalized-split}
 T_{n,\alpha}^{\RR}(u)
 =X_n(u)+Y_{n,p,\alpha}(u),
\end{equation}
где
\begin{align}\label{eq:normalized-error-part}
 Y_{n,p,\alpha}(u)
 :=-\frac{\MKQ\widehat g(Z_{n/2})-\MKQ\widehat g(Z_n)}
 {\sqrt{n/2}\,\sigma(u)}\notag +\frac{\sqrt{n/2}}{\sigma(u)}
 v^\transpose\scrE_{n,p,\alpha}^{\RR}.
\end{align}
Оценки \eqref{eq:levin-remainder} и \eqref{eq:direction-uniform-bounds},
а также ограниченность решения уравнения Пуассона дают
\[
 \left\{\PE_\xi[\abs{Y_{n,p,\alpha}(u)}^p]\right\}^{1/p}
 \leq\mathcal R_{n,p,\alpha}
\]
для любого начального распределения $\xi$. Применяя
\eqref{eq:smoothing-inequality} к $X=X_n(u)$ и
$Y=Y_{n,p,\alpha}(u)$, и используя Лемму~\ref{lem:martingale-gaussian}, получаем
\begin{align*}
 \dK\!\left(T_{n,\alpha}^{\RR}(u),\gauss(0,1)\right)
 &\leq\dK\!\left(X_n(u),\gauss(0,1)\right)
   +2\bigl\{\PE_\xi[\abs{Y_{n,p,\alpha}(u)}^p]\bigr\}^{1/(p+1)}\\
 &\leq\mathcal B_{\rm M}(n,p)
   +2\bigl\{\mathcal R_{n,p,\alpha}\bigr\}^{p/(p+1)}.
\end{align*}
Это доказывает \eqref{eq:intermediate-gaussian}.
\end{proof}

\begin{theorem}[Гауссовская аппроксимация]
\label{thm:gaussian}
Пусть выполнены условия 1--3 и $n$ чётно. Предположим также, что
$\noisecovmarkov\succ0$, и зафиксируем $u\in\sphere^{d-1}$.
Если $n$ достаточно велико и $0<\alpha\leq\alpha_n^{\rm L}$,
то
\begin{equation}\label{eq:main-gaussian}
 \dK\!\left(
  \frac{\sqrt{n/2}\,u^\transpose
   (\btheta_n^{(\alpha,\RR)}-\thetalim)}{\sigma(u)},
  \gauss(0,1)\right)
 \lesssim_{\log{n}} C_{\rm G}(d)\,r_{n,\alpha}.
\end{equation}
Здесь $C_{\rm G}(d)\geq1$ не зависит от $n$, $\alpha$ и $u$.
Оценка равномерна по $u\in\sphere^{d-1}$ и начальному распределению $\xi$.
\end{theorem}

\begin{proof}
Положим
\[
 p=p_n:=\max\{2,\ceil{\log n}\}.
\]
По определению
$\alpha_n^{\rm L}=\alpha_p^{\rm L}$. Введём используемый только в доказательстве
нормированный масштаб остаточного члена
\begin{equation}\label{eq:rho-rate}
 \rho_{n,p,\alpha}:=\sqrt{n/2}\,r_{n,p,\alpha}.
\end{equation}
Из \eqref{eq:levin-rate} следует
\begin{align}\label{eq:rho-rate-expanded}
 \rho_{n,p,\alpha}
 &\lesssim p\log(2p)n^{-1/4}
 +\frac{p^3L_\alpha^{1/2}}{\sqrt{\alpha n}}
 +p^{3/2}\sqrt\alpha
 +p^{7/2}\sqrt n\,\alpha^{3/2}L_\alpha^{3/2}\notag\\
 &\quad
 +p^{7/2}n^{-1/2}\alpha^{1/2}L_\alpha^{3/2}
 +\frac{\norm{\thetainit-\thetalim}}{\alpha\sqrt n}
   \exp\!\left(-\frac{\alpha an}{24}\right).
\end{align}
Следовательно,
\begin{equation}\label{eq:rho-log-bound}
 \rho_{n,p,\alpha}
 \lesssim_{\log{n}}r_{n,\alpha} \eqsp,
\end{equation}
где $r_{n,\alpha}$ определяется в \eqref{eq:alpha-n-rate}.

Применим Лемму~\ref{lem:rr-remainder} при $p=p_n$. Поскольку
$p\geq\log n$,
из \eqref{eq:BM-definition}, \eqref{eq:rho-log-bound} и
\eqref{eq:RR-normalized-remainder} следуют оценки
\begin{equation}\label{eq:log-moment-rates}
 \mathcal B_{\rm M}(n,p)
 \lesssim_{\log{n}} C_{\rm G}(d)n^{-1/4},
 \qquad
 \mathcal R_{n,p,\alpha}
 \lesssim_{\log{n}} C_{\rm G}(d)\,r_{n,\alpha}.
\end{equation}
Степень $p/(p+1)$ в \eqref{eq:intermediate-gaussian} также можно убрать из
окончательной формулировки. Если
$\mathcal R_{n,p,\alpha}\geq n^{-1/4}$, то
\[
 \mathcal R_{n,p,\alpha}^{p/(p+1)}
 \leq e^{1/4}\mathcal R_{n,p,\alpha};
\]
если же $\mathcal R_{n,p,\alpha}<n^{-1/4}$, то
\[
 \mathcal R_{n,p,\alpha}^{p/(p+1)}
 \leq n^{-p/(4(p+1))}
 \leq e^{1/4}n^{-1/4}.
\]
Подстановка этих двух оценок в \eqref{eq:intermediate-gaussian} даёт
\eqref{eq:main-gaussian}.
\end{proof}

Формула \eqref{eq:main-gaussian} позволяет выбирать шаг, не фиксируя его
заранее. При $\alpha=c/\sqrt n$ с фиксированным $c>0$ получаем
$\dK\lesssim_{\log{n}}C_{\rm G}(d)n^{-1/4}$, а переходный член
экспоненциально мал. Общая формула \eqref{eq:alpha-n-rate} показывает цену
слишком малого шага через $(\alpha n)^{-1/2}L_\alpha^{1/2}$ и переходный
член, а слишком большого --- через $\sqrt\alpha$ и
$\sqrt n\,\alpha^{3/2}L_\alpha^{3/2}$.
