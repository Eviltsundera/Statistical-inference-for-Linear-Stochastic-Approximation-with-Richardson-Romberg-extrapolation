
\section{Summed Bounds and Comparison with the Single-Step Case}

\begin{corollary}\label{cor:RR-weight-summed}
  \textbf{(Summed RR weight bounds.)}
  Under the assumptions of the previous lemma, uniformly in \(n \ge 2\),
\begin{equation}
\begin{aligned}
  \sum_{l = 1}^{n - 1} \lVert  \mathcal{Q}_l^{\text{RR}} - \overline{A}^{-1}  \rVert^2
    \le \frac{C_1}{\alpha a},
  \quad
  \sum_{l = 1}^{n - 2} \lVert  \mathcal{Q}_{l + 1}^{\text{RR}} - \mathcal{Q}_l^{\text{RR}}  \rVert
    \le \frac{C_2}{a^2},
\end{aligned}
\end{equation}
  with \(C_1 = 9 C_Q^2\) and \(C_2 = 32 \widetilde{C}_A\).
\end{corollary}
