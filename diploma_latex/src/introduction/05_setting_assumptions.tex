\section{Setting and assumptions}
\label{sec:assumptions}

We now formalize the setting and state the assumptions that will be used
throughout this work.

Let $\{Z_k\}_{k\in\mathbb{N}}$ be a Markov chain on a complete separable
metric space $(\mathsf{Z},\mathcal{Z})$ with transition kernel $\mathsf{Q}$.

\begin{assumption}[Uniform geometric ergodicity]
  The kernel $\mathsf{Q}$ admits a unique invariant distribution $\pi$ and is
  \emph{uniformly geometrically ergodic}: there exists
  $t_{\mathrm{mix}}\in\mathbb{N}^*$ such that for all
  $k\in\mathbb{N}^*$,
  \[
    \Delta(\mathsf{Q}^k)
      :=\sup_{z,z'\in\mathsf{Z}}\frac{1}{2}
        \bigl\lVert
          \mathsf{Q}^k(z,\mathord{\cdot})
          -\mathsf{Q}^k(z',\mathord{\cdot})
        \bigr\rVert_{\mathrm{TV}}
      \leq \left(\frac{1}{4}\right)^{\lfloor k/t_{\mathrm{mix}}\rfloor}.
  \]
  Equivalently, there exist constants $\zeta>0$ and $\rho\in(0,1)$ such that
  \[
    \sup_z\bigl\lVert
      \mathsf{Q}^k(z,\mathord{\cdot})-\pi
    \bigr\rVert_{\mathrm{TV}}
      \leq\zeta\rho^k
  \]
  for all $k\geq1$.
\end{assumption}

\begin{assumption}[Hurwitz condition and boundedness]
  The matrix $-\overline{A}$ is Hurwitz; equivalently, all eigenvalues of
  $\overline{A}$ have strictly positive real parts. Moreover,
  \[
    C_A:=\max\left\lbrace
      \sup_{z\in\mathsf{Z}}\lVert A(z)\rVert,
      \sup_{z\in\mathsf{Z}}\lVert\widetilde{A}(z)\rVert
    \right\rbrace<\infty,
  \]
  where $\widetilde{A}(z):=A(z)-\overline{A}$.
\end{assumption}

\begin{assumption}[Noise regularity]
  The noise function
  $\varepsilon(z)=\widetilde{A}(z)\theta^*-\widetilde{b}(z)$, where
  $\widetilde{b}(z)=b(z)-\overline{b}$, satisfies
  \[
    \lVert\varepsilon\rVert_\infty
      :=\sup_{z\in\mathsf{Z}}\lVert\varepsilon(z)\rVert<+\infty.
  \]
\end{assumption}

By construction, $\pi(\widetilde{A})=0$, $\pi(\widetilde{b})=0$, and hence
$\pi(\varepsilon)=0$.

Under Assumptions~1--3, the error $\theta_k^{(\alpha)}-\theta^*$ satisfies
the recursion
\begin{equation}
  \theta_k^{(\alpha)}-\theta^*
  =\bigl(I-\alpha A(Z_k)\bigr)
    \bigl(\theta_{k-1}^{(\alpha)}-\theta^*\bigr)
    -\alpha\varepsilon(Z_k).
  \label{eq:error-recursion}
\end{equation}
