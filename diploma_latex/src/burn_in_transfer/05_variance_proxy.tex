
\section{Burned-in Variance Proxy}

For the stochastic depth-zero term, write
$Q_{l}^{\mathrm{bRR}} := Q_{l; n_{0}, n}^{\mathrm{RR}}$ and
$\Sigma := \Sigma_{\varepsilon}^{(\mathrm{M})}$. We take $n \ge 2$ in this subsection.
The martingale part produced by the Poisson decomposition starts at $l = 2$, so
the deterministic variance proxy is
\begin{equation}
\Sigma_{n,n_{0}}^{\mathrm{bRR}}
  := \frac{1}{m} \sum_{l = 2}^{n - 1}
      Q_{l}^{\mathrm{bRR}} \, \Sigma \, (Q_{l}^{\mathrm{bRR}})^{\mathsf{T}},
\quad
\sigma_{n,n_{0}}^{2, \mathrm{bRR}}(u)
  := u^{\mathsf{T}} \Sigma_{n,n_{0}}^{\mathrm{bRR}} u.
\label{eq:burn-variance-proxy}
\end{equation}

\begin{lemma}\label{lem:burn-variance-comparison}
  \textbf{(Burned-in variance comparison.)}
  Assume the conditions of \cref{lem:burn-rr-weight-bounds} and $\lVert \Sigma \rVert < \infty$.
  There exists a constant $C_{\mathrm{burn,3}}$, depending only on
  $\kappa_{Q}$, $\lVert \overline{A}^{-1} \rVert$, $\lVert \overline{A} \rVert$, and $\lVert \Sigma \rVert$,
  such that
  \begin{equation}
\lVert \Sigma_{n,n_{0}}^{\mathrm{bRR}} - \Sigma_{\infty} \rVert
    \le \frac{C_{\mathrm{burn,3}}}{m \, \alpha a}.
\label{eq:burn-variance-comparison}
\end{equation}
  Consequently,
  \begin{equation}
| \sigma_{n,n_{0}}^{2, \mathrm{bRR}}(u) - \sigma^{2}(u) |
    \le \frac{C_{\mathrm{burn,3}} \lVert u \rVert^{2}}{m \, \alpha a}.
\label{eq:burn-scalar-variance-comparison}
\end{equation}
\end{lemma}

\begin{proof} Put
\[
I_{\mathrm{post}} := \left\{ l: 2 \le l \le n - 1, l \ge n_{0} \right\},
\quad
I_{\mathrm{pre}} := \left\{ l: 2 \le l < n_{0} \right\},
\quad
r_{m} := m - |I_{\mathrm{post}}| \in \left\{ 0,1,2 \right\}.
\]
For $l \in I_{\mathrm{post}}$, set
$\Delta_{l} := Q_{l}^{\mathrm{bRR}} - \overline{A}^{-1}$. Then
\[
Q_{l}^{\mathrm{bRR}} \Sigma (Q_{l}^{\mathrm{bRR}})^{\mathsf{T}} - \Sigma_{\infty}
  = \Delta_{l} \Sigma \overline{A}^{-\mathsf{T}}
    + \overline{A}^{-1} \Sigma \Delta_{l}^{\mathsf{T}}
    + \Delta_{l} \Sigma \Delta_{l}^{\mathsf{T}}.
\]
Therefore
\[
\begin{aligned}
\Sigma_{n,n_{0}}^{\mathrm{bRR}} - \Sigma_{\infty}
  &= \frac{1}{m} \sum_{l \in I_{\mathrm{post}}}
      \left( Q_{l}^{\mathrm{bRR}} \Sigma (Q_{l}^{\mathrm{bRR}})^{\mathsf{T}} - \Sigma_{\infty} \right)
  &\quad + \frac{1}{m} \sum_{l \in I_{\mathrm{pre}}}
      Q_{l}^{\mathrm{bRR}} \Sigma (Q_{l}^{\mathrm{bRR}})^{\mathsf{T}}
      - \frac{r_{m}}{m} \Sigma_{\infty}.
\end{aligned}
\]

The post-burn-in contribution is bounded by
\[
\begin{aligned}
&\sum_{l \in I_{\mathrm{post}}}
  \lVert Q_{l}^{\mathrm{bRR}} \Sigma (Q_{l}^{\mathrm{bRR}})^{\mathsf{T}} - \Sigma_{\infty} \rVert
&\quad \le
  2 \lVert \overline{A}^{-1} \rVert \lVert \Sigma \rVert \sum_{l \in I_{\mathrm{post}}} \lVert \Delta_{l} \rVert
  + \lVert \Sigma \rVert \sum_{l \in I_{\mathrm{post}}} \lVert \Delta_{l} \rVert^{2}
&\quad \le \frac{C}{\alpha a},
\end{aligned}
\]
where \eqref{eq:burn-post-weight-error} gives the first geometric sum and
\eqref{eq:burn-weight-energy} gives the squared sum. For the pre-burn-in part,
\eqref{eq:burn-weight-energy} gives
\[
\sum_{l \in I_{\mathrm{pre}}}
  \lVert Q_{l}^{\mathrm{bRR}} \Sigma (Q_{l}^{\mathrm{bRR}})^{\mathsf{T}} \rVert
  \le \lVert \Sigma \rVert \sum_{l \in I_{\mathrm{pre}}} \lVert Q_{l}^{\mathrm{bRR}} \rVert^{2}
  \le \frac{C}{\alpha a}.
\]
Finally,
\[
\frac{r_{m}}{m} \lVert \Sigma_{\infty} \rVert
  \le \frac{2 \lVert \overline{A}^{-1} \rVert^{2} \lVert \Sigma \rVert}{m}
  \le \frac{C}{m \alpha a},
\]
using $\alpha a \le 1$. Combining the last three displays and dividing by $m$
gives \eqref{eq:burn-variance-comparison}. Also,
\[
|\sigma_{n,n_{0}}^{2, \mathrm{bRR}}(u) - \sigma^{2}(u)|
  = |u^{\mathsf{T}} (\Sigma_{n,n_{0}}^{\mathrm{bRR}} - \Sigma_{\infty}) u|
  \le \lVert \Sigma_{n,n_{0}}^{\mathrm{bRR}} - \Sigma_{\infty} \rVert \lVert u \rVert^{2},
\]
which gives \eqref{eq:burn-scalar-variance-comparison}. \end{proof}

Assume $\sigma^{2}(u) > 0$ and impose the burned-in variance lower-bound condition
\begin{equation}
m \, \alpha \, a
  \ge \frac{2 C_{\mathrm{burn,3}} \lVert u \rVert^{2}}{\sigma^{2}(u)}.
\label{eq:burn-variance-lb-condition}
\end{equation}
Then \eqref{eq:burn-scalar-variance-comparison} gives
$\sigma_{n,n_{0}}^{2, \mathrm{bRR}}(u) \ge \sigma^{2}(u) / 2$, hence
$\sigma_{n,n_{0}}^{\mathrm{bRR}}(u) \ge \sigma(u) / \sqrt{2}$.
% At the balanced scale $\alpha = c \, n^{-1 / 2}$ with $m \ge n / 2$,
% this condition holds for all sufficiently large $n$ depending on $u$, $c$,
% and the problem constants.

We say that $(n,n_{0},p,q,\alpha,u)$ is in the \emph{admissible burn-in regime} if
$m := n - n_{0}$ and
\[
n \ge 3,
\quad
0 \le n_{0} < n,
\quad
p \ge 2,
\quad
q \ge 2,
\quad
p \le q / 4,
\quad
\sigma^{2}(u) > 0,
\]
\begin{equation}
m \ge n / 2,
\quad
0 < \alpha,
\quad
2 \alpha \le \alpha_{\mathrm{burn}}(p,q),
\quad
m \, \alpha \, a
  \ge \frac{2 C_{\mathrm{burn,3}} \lVert u \rVert^{2}}{\sigma^{2}(u)}.
\label{eq:admissible-burn-regime}
\end{equation}
The last inequality is \eqref{eq:burn-variance-lb-condition}.
